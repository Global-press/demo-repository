🔧 Deployment Flow
# GitHub Action: Auto-publish news
on:
  schedule:
    - cron: '0 * * * *' # hourly
jobs:
  fetch-news:
    runs-on: ubuntu-latest
    steps:
      - name: Fetch RSS
        run: curl https://newsapi.org/v2/top-headlines?... > news.json
      - name: Format HTML
        run: node format-news.js
      - name: Commit & Push
        run: git commit -am "Auto-update news" && git push
